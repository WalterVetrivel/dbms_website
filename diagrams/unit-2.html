GROUP BY with HAVING

WHERE filters rows, GROUP BY makes groups and HAVING filters the groups.

SQL 
SELECT COMPANY, COUNT(*) AS items, SUM(COST) AS total
FROM PRODUCT_MAST
WHERE RATE >= 20
GROUP BY COMPANY
HAVING SUM(COST) > 150
ORDER BY total DESC;
Output 
+---------+-------+-------+
| COMPANY | items | total |
+---------+-------+-------+
| Com1    |     3 |   285 |
| Com2    |     3 |   165 |
+---------+-------+-------+

Inner join

Only the students whose department matches a row in department.

SQL 
SELECT s.name, d.dept_name
FROM student s
INNER JOIN department d ON s.dept_id = d.dept_id
ORDER BY s.roll;
Output 
+---------+-----------+
| name    | dept_name |
+---------+-----------+
| Anitha  | CSE       |
| Bharath | IT        |
| Charan  | CSE       |
+---------+-----------+

Left outer join

Every student is kept. A student with no department gets NULL.

SQL 
SELECT s.name, d.dept_name
FROM student s
LEFT JOIN department d ON s.dept_id = d.dept_id
ORDER BY s.roll;
Output 
+---------+-----------+
| name    | dept_name |
+---------+-----------+
| Anitha  | CSE       |
| Bharath | IT        |
| Charan  | CSE       |
| Divya   | NULL      |
+---------+-----------+

Self join

Each employee next to their manager, from one table.

SQL 
CREATE TABLE employee (
  emp_id     INT PRIMARY KEY,
  name       VARCHAR(20),
  manager_id INT
);
INSERT INTO employee VALUES (1, 'Ravi', NULL), (2, 'Meena', 1), (3, 'Kumar', 1), (4, 'Asha', 2);

SELECT e.name AS employee, m.name AS manager
FROM employee e
LEFT JOIN employee m ON e.manager_id = m.emp_id
ORDER BY e.emp_id;
Output 
+----------+---------+
| employee | manager |
+----------+---------+
| Ravi     | NULL    |
| Meena    | Ravi    |
| Kumar    | Ravi    |
| Asha     | Meena   |
+----------+---------+

Correlated subquery

Employees who earn more than the average of their own department.

SQL 
SELECT NAME, DEPT, SALARY
FROM EMPLOYEE e
WHERE SALARY > (SELECT AVG(SALARY)
                FROM EMPLOYEE
                WHERE DEPT = e.DEPT);
Output 
+---------+-------+----------+
| NAME    | DEPT  | SALARY   |
+---------+-------+----------+
| John    | Sales |  2000.00 |
| Alina   | IT    |  6500.00 |
| Kathrin | HR    |  8500.00 |
| Jackson | IT    | 10000.00 |
+---------+-------+----------+

Trigger: an audit log

AFTER INSERT, UPDATE and DELETE triggers that log every change to an account, with the old and new balance.

SQL 
CREATE TABLE balance_log (
  log_id      INT AUTO_INCREMENT PRIMARY KEY,
  acc_no      INT,
  action      VARCHAR(10),
  old_balance DECIMAL(10,2),
  new_balance DECIMAL(10,2),
  logged_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TRIGGER account_ai AFTER INSERT ON account
FOR EACH ROW
INSERT INTO balance_log (acc_no, action, old_balance, new_balance)
VALUES (NEW.acc_no, 'INSERT', NULL, NEW.balance);

CREATE TRIGGER account_au AFTER UPDATE ON account
FOR EACH ROW
INSERT INTO balance_log (acc_no, action, old_balance, new_balance)
VALUES (NEW.acc_no, 'UPDATE', OLD.balance, NEW.balance);

CREATE TRIGGER account_ad AFTER DELETE ON account
FOR EACH ROW
INSERT INTO balance_log (acc_no, action, old_balance, new_balance)
VALUES (OLD.acc_no, 'DELETE', OLD.balance, NULL);
SQL 
INSERT INTO account VALUES (104, 'Divya', 500);
UPDATE account SET balance = balance - 200 WHERE acc_no = 101;
DELETE FROM account WHERE acc_no = 102;

SELECT log_id, acc_no, action, old_balance, new_balance FROM balance_log;
Output 
+--------+--------+--------+-------------+-------------+
| log_id | acc_no | action | old_balance | new_balance |
+--------+--------+--------+-------------+-------------+
|      1 |    104 | INSERT |        NULL |      500.00 |
|      2 |    101 | UPDATE |     1000.00 |      800.00 |
|      3 |    102 | DELETE |      300.00 |        NULL |
+--------+--------+--------+-------------+-------------+

Attribute closure

The closure algorithm, and A+ worked out one FD at a time.

Algorithm 
result := X
repeat
  for each FD  Y → Z  in F do
    if Y ⊆ result then result := result ∪ Z
until result does not change
return result
